- [ ] **Step 1: 스크립트를 쓴다**

```bash
cat > $SCRATCH/audit7_trace.py <<'PY'
import re, subprocess, sys, os, io, pathlib

ROOT = "/home/pi/sop-project"
BASE = os.environ.get("BASE", "0d19f6c")
BASE5 = os.environ.get("BASE5", "e9483e3")

# 대조 대상: 규칙이 빠져나간 문서
SRC = ["CLAUDE.md", "README.md", "docs/claude-code-작업문서.md", "docs/통합문서.md"]

def deleted_lines(repo, base, paths):
    out = subprocess.run(["git","-C",repo,"-c","core.quotepath=false","diff",
                          f"{base}..HEAD","--"]+paths,
                         capture_output=True, text=True).stdout
    res=[]
    cur=None
    for ln in out.splitlines():
        if ln.startswith("+++ b/"): cur = ln[6:]
        if ln.startswith("-") and not ln.startswith("---"):
            t = ln[1:]
            if not t.strip(): continue
            if re.fullmatch(r"[\s|:\-]+", t): continue   # 표 구분선
            res.append((cur, t))
    return res

# 키 어구: 백틱 안 식별자 우선, 없으면 한글 6자 이상 연속
def key_of(text):
    m = re.findall(r"`([^`]{3,})`", text)
    if m: return max(m, key=len)
    m = re.findall(r"[가-힣]{6,}", text)
    if m: return max(m, key=len)
    m = re.findall(r"[A-Za-z0-9_./-]{6,}", text)
    return max(m, key=len) if m else None

# 새 자리 = 신설·수정된 모든 면
def haystack():
    parts=[]
    for p in pathlib.Path(ROOT).glob(".claude/**/*.md"):
        parts.append(p.read_text(encoding="utf-8"))
    for p in [".claude/settings.json","CLAUDE.md","README.md",
              "docs/claude-code-작업문서.md","docs/통합문서.md",
              "docs/변경이력.md","Rpi5/CLAUDE.md"]:
        f = pathlib.Path(ROOT)/p
        if f.exists(): parts.append(f.read_text(encoding="utf-8"))
    hd = pathlib.Path.home()/"hanium-docs/.claude/rules"
    if hd.exists():
        for p in hd.glob("*.md"): parts.append(p.read_text(encoding="utf-8"))
    return "\n".join(parts)

# 🔴 원장은 원자화하며 원문을 그대로 인용했다 — 단순히 '원장에 있다'는 근거가 못 된다.
#    그 줄이 실린 원장 행의 '판정'이 제거일 때만 '버림(근거 있음)'이다.
LEDGER_ROWS = [l for l in (pathlib.Path(ROOT)/"docs/superpowers/plans/2026-09-20-규칙표면-원장.md")
               .read_text(encoding="utf-8").splitlines() if l.startswith("|")]

def ledger_verdict(k):
    hit = [l for l in LEDGER_ROWS if k in l]
    if not hit: return None
    return "제거" if any("제거" in l for l in hit) else "판정있음(제거아님)"

def main():
    H = haystack()
    rows = deleted_lines(ROOT, BASE, SRC) + deleted_lines(ROOT+"/Rpi5", BASE5, ["CLAUDE.md"])
    counts={"이사함":0,"버림(근거 있음)":0,"행방불명":0,"키없음":0}
    with io.open(os.environ["SCRATCH"]+"/trace_table.tsv","w",encoding="utf-8") as fo:
        fo.write("분류\t파일\t키\t원문\n")
        for f,t in rows:
            k = key_of(t)
            if not k:
                cls="키없음"
            elif k in H:
                cls="이사함"
            elif ledger_verdict(k) == "제거":
                cls="버림(근거 있음)"
            else:
                cls="행방불명"
            counts[cls]+=1
            fo.write(f"{cls}\t{f}\t{k}\t{t[:160]}\n")
    for k,v in counts.items(): print(f"{k}\t{v}")
    print("합계\t", sum(counts.values()))

main()
PY
```
