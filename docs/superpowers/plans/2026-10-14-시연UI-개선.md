```python
# 시안(설계 2026-10-13 §2.1)은 영상 칸 1707×1280 을 기준으로 그렸다 — 위치 · 크기는 그 px 에 (영상 높이 ÷ 1280)을 곱한다.
REF_H = 1280


def vpx(vr, px):
    """시안 px → 지금 영상 사각형의 px."""
    return int(round(px * vr.height() / REF_H))


def vplace(widget, vr, x, y, w=None, h=None):
    """영상 사각형 원점 기준 시안 좌표(px)로 앉힌다. w·h 가 None 이면 sizeHint(폭이 정해졌으면 그 폭에서의 높이)."""
    ww = vpx(vr, w) if w is not None else widget.sizeHint().width()
    if h is not None:
        hh = vpx(vr, h)
    elif widget.hasHeightForWidth():
        hh = widget.heightForWidth(ww)
    else:
        hh = widget.sizeHint().height()
    widget.setGeometry(QRect(vr.left() + vpx(vr, x), vr.top() + vpx(vr, y), ww, hh))
```

```python
from PyQt6.QtCore import QPointF, QRectF
from PyQt6.QtGui import QPainter, QPainterPath, QColor, QPen


def cut_corner_path(rect, c):
    """왼쪽 위 · 오른쪽 아래 모서리를 c 만큼 자른 다각형(시안 clip-path) — 효과가 아니라 도형 한 번."""
    r = QRectF(rect)
    p = QPainterPath()
    p.moveTo(r.left() + c, r.top()); p.lineTo(r.right(), r.top())
    p.lineTo(r.right(), r.bottom() - c); p.lineTo(r.right() - c, r.bottom())
    p.lineTo(r.left(), r.bottom()); p.lineTo(r.left(), r.top() + c); p.closeSubpath()
    return p


def _qcolor(css):
    """'#rrggbb' · 'rgba(r, g, b, a)' → QColor."""
    if css.startswith("rgba"):
        r, g, b, a = [x.strip() for x in css[css.index("(") + 1:-1].split(",")]
        return QColor(int(r), int(g), int(b), int(float(a) * 255))
    return QColor(css)


class ConnBar(QWidget):
    """우하단 상태 막대 — 카메라 ● fps │ 인터락 ● │ 버튼(GPIO) ●(설계 §3 ⑥).
    🔴 상태 판단은 하지 않는다 — 콘솔이 점검과 같은 출처로 넣는다. 그리기는 값이 바뀔 때만(update)."""

    ITEMS = (("camera", "카메라"), ("interlock", "인터락"), ("gpio", "버튼"))

    def __init__(self, parent=None):
        super().__init__(parent)
        self.setAttribute(Qt.WidgetAttribute.WA_TransparentForMouseEvents, True)
        self.setFont(config.font("sub", 600))
        self._states = {k: None for k, _ in self.ITEMS}
        self._fps = None

    def _color(self, key):
        v = self._states.get(key)
        return theme.C("done" if v is True else "danger" if v is False else "todo")

    def update_state(self, states, fps=None):
        new = dict(self._states); new.update(states or {})
        if new != self._states or fps != self._fps:
            self._states, self._fps = new, fps
            self.update()

    def text(self):
        parts = []
        for key, name in self.ITEMS:
            parts.append(f"{name} ●" + (f" {self._fps:.1f} fps" if key == "camera" and self._fps is not None else ""))
        return " │ ".join(parts)

    def paintEvent(self, _event):
        p = QPainter(self); p.setRenderHint(QPainter.RenderHint.Antialiasing)
        p.fillPath(cut_corner_path(self.rect().adjusted(0, 0, -1, -1), 10), _qcolor(theme.C("panel_bg")))
        fm = p.fontMetrics(); x = 16; y = (self.height() + fm.ascent() - fm.descent()) // 2
        for i, (key, name) in enumerate(self.ITEMS):
            if i:
                p.setPen(_qcolor(theme.C("panel_border"))); p.drawText(x, y, "│"); x += fm.horizontalAdvance("│") + 12
            p.setPen(_qcolor(theme.C("text"))); p.drawText(x, y, name); x += fm.horizontalAdvance(name) + 8
            p.setPen(Qt.PenStyle.NoPen); p.setBrush(_qcolor(self._color(key)))
            p.drawEllipse(QPointF(x + 6, self.height() / 2), 6, 6); x += 20
            if key == "camera" and self._fps is not None:
                s = f"{self._fps:.1f} fps"; p.setPen(_qcolor(theme.C("label"))); p.drawText(x, y, s)
                x += fm.horizontalAdvance(s) + 12
        p.end()

    def apply_theme(self):
        self.update()

    def relayout(self, vr):
        vplace(self, vr, 1294, 1204, 380, 44)
```

```python
    """🔴 16진수 → BGR 는 여기 하나뿐이다(cv2 는 BGR)."""

    return (int(h[4:6], 16), int(h[2:4], 16), int(h[0:2], 16))

def role_bgr(role):

def _glyph(frame, role, x, y, s):
    """이름표 앞 기호 — Hershey 에 ✓ ✕ 자물쇠가 없어 선으로(설계 §3.1)."""
    if role == "done":

        cv2.line(frame, (x + s // 3, y + s), (x + s, y), _INK, 2)
    elif role == "warn":
        cv2.line(frame, (x, y), (x + s, y + s), _INK, 2)
        cv2.line(frame, (x + s, y), (x, y + s), _INK, 2)
    elif role == "block":
        cv2.rectangle(frame, (x, y + s // 2), (x + s, y + s), _INK, -1)
        cv2.ellipse(frame, (x + s // 2, y + s // 2), (s // 3, s // 3), 0, 180, 360, _INK, 2)
    else:
        return 0
    return s + 6


def draw_box_label(frame, box, name, role, score=None):
    """모서리 꺾쇠 8줄 + 작은 이름표(설계 §3 ②). 위쪽 끝 상자는 이름표를 상자 안으로 내린다."""
    x1, y1, x2, y2 = (int(v) for v in box)
    c = role_bgr(role)
    L = max(8, min(x2 - x1, y2 - y1) // 4)
    for (px, py, dx, dy) in ((x1, y1, 1, 1), (x2, y1, -1, 1), (x1, y2, 1, -1), (x2, y2, -1, -1)):
        cv2.line(frame, (px, py), (px + dx * L, py), c, 3)
        cv2.line(frame, (px, py), (px, py + dy * L), c, 3)
    text = name if score is None else f"{name} {score:.2f}"
    (tw, th), base = cv2.getTextSize(text, cv2.FONT_HERSHEY_SIMPLEX, 0.6, 2)
    gs = th if role in ("done", "warn", "block") else 0
    w = tw + 12 + (gs + 6 if gs else 0); h = th + base + 8
    top = y1 - h - 2 if y1 - h - 2 >= 0 else y1 + 2
    cv2.rectangle(frame, (x1, top), (x1 + w, top + h), c, -1)
    gx = _glyph(frame, role, x1 + 6, top + 4, gs) if gs else 0
    cv2.putText(frame, text, (x1 + 6 + gx, top + h - base - 3), cv2.FONT_HERSHEY_SIMPLEX, 0.6, _INK, 2)
```

```python
class ConfirmPanel(_Sheet):
    """앱 안 확인 상자(설계 2026-10-13 §4.6) — 운영체제 창(QMessageBox) 대신. 🔴 기본 = 취소(오동작 방지 · 종전과 같다)."""

    accepted = pyqtSignal()
    closed = pyqtSignal()

    def __init__(self, title, body, ok_text, cancel_text, parent=None):
        super().__init__(parent)
        lay = QVBoxLayout(self); lay.setContentsMargins(40, 36, 40, 32); lay.setSpacing(16)
        self._title = QLabel(title); self._title.setFont(config.font("sheet_title", 800))
        self._body = QLabel(body); self._body.setFont(config.font("sub", 500)); self._body.setWordWrap(True)
        lay.addWidget(self._title); lay.addWidget(self._body)
        row = QHBoxLayout(); row.setSpacing(16)
        self.cancel_button = QPushButton(cancel_text); self.ok_button = QPushButton(ok_text)
        for b in (self.cancel_button, self.ok_button):
            b.setFont(config.font("item", 800)); b.setMinimumHeight(72)
            b.setCursor(Qt.CursorShape.PointingHandCursor); row.addWidget(b, 1)
        self.cancel_button.setDefault(True)
        self.cancel_button.clicked.connect(self.closed.emit)
        self.ok_button.clicked.connect(self.accepted.emit)
        lay.addLayout(row)

    def apply_theme(self):
        super().apply_theme()
        self._title.setStyleSheet(theme.text_qss("text", 800))
        self._body.setStyleSheet(theme.text_qss("label", 500))
        self.cancel_button.setStyleSheet(f"QPushButton {{ background: {theme.C('current')}; color: #111111; border: none; }}")
        self.ok_button.setStyleSheet(f"QPushButton {{ background: transparent; color: {theme.C('warn_text')};"
                                     f" border: 2px solid {theme.C('warn')}; }}")

    def relayout(self, vr):
        w = vpx(vr, 760); self.setFixedWidth(w)
        h = self.layout().heightForWidth(w) if self.layout().hasHeightForWidth() else self.sizeHint().height()
        self.setGeometry(QRect(vr.left() + (vr.width() - w) // 2, vr.top() + (vr.height() - h) // 2, w, h))
```
