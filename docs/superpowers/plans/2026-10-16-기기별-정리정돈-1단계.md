기록 = Path(__file__).resolve().with_name("옮김기록.tsv")
열 = ["번호", "시각", "방식", "원래", "새", "빼기", "파일수", "바이트"]


class 멈춤(Exception):
    """이 줄을 하지 않고(또는 하다가 치우고) 멈춘다."""


def md5(p):
    h = hashlib.md5()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def 같은장치(a, b):
    """b 는 아직 없을 수 있다 — 있는 가장 가까운 조상으로 본다."""
    b = Path(b)
    while not b.exists():
        b = b.parent
    return os.lstat(a).st_dev == os.stat(b).st_dev


def 자리(root, rel):
    return str(root) if rel == "." else os.path.join(root, rel)


def 빠짐(rel, 빼기):
    return any(rel == x or rel.startswith(x + "/") for x in 빼기)


def 훑기(root, 빼기):
    """[(상대경로, 종류)] — 종류 = 파일 · 링크 · 폴더. root 가 파일 · 링크면 [('.', 종류)]."""
    m = os.lstat(root).st_mode
    if stat.S_ISLNK(m):
        return [(".", "링크")]
    if stat.S_ISREG(m):
        return [(".", "파일")]
    if not stat.S_ISDIR(m):
        raise 멈춤(f"특수 파일: {root}")
    out = []
    for dp, dns, fns in os.walk(root):
        아래 = []
        for n in sorted(dns) + sorted(fns):
            p = os.path.join(dp, n)
            rel = os.path.relpath(p, root)
            if 빠짐(rel, 빼기):
                continue
            k = os.lstat(p).st_mode
            if stat.S_ISLNK(k):
                out.append((rel, "링크"))
            elif stat.S_ISDIR(k):
                out.append((rel, "폴더"))
                아래.append(n)
            elif stat.S_ISREG(k):
                out.append((rel, "파일"))
            else:
                raise 멈춤(f"특수 파일: {p}")
        dns[:] = [d for d in dns if d in 아래]
    return out


def 지문(p, 종류):
    if 종류 == "링크":
        return "→" + os.readlink(p) if os.path.islink(p) else "?"
    if 종류 == "폴더":
        return "폴더" if os.path.isdir(p) and not os.path.islink(p) else "?"
    if not os.path.isfile(p) or os.path.islink(p):
        return "?"
    return md5(p)


def 확인(src, dst, 합침=False):
    if not os.path.lexists(src):
        raise 멈춤(f"원래 자리가 없다: {src}")
    if os.path.lexists(dst) and not 합침:
        raise 멈춤(f"새 자리가 이미 있다(덮어쓰지 않음): {dst}")
    s, d = os.path.realpath(src), os.path.realpath(dst)
    if d == s or d.startswith(s + "/") or s.startswith(d + "/"):
        raise 멈춤(f"원래 자리와 새 자리가 겹친다: {src} → {dst}")


def 세기(root, 목록):
    return (sum(1 for _, k in 목록 if k != "폴더"),
            sum(os.lstat(자리(root, r)).st_size for r, k in 목록 if k == "파일"))


def 이름바꾸기(src, dst, 빼기):
    if 빼기:
        raise 멈춤("이름바꾸기에는 빼기를 쓰지 않는다 — 복사이동으로")
    확인(src, dst)
    if not 같은장치(src, os.path.dirname(dst)):
        raise 멈춤(f"다른 파일 시스템 — 복사이동으로: {src} → {dst}")
    n, b = 세기(src, 훑기(src, []))
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    os.rename(src, dst)
    return n, b


def 복사이동(src, dst, 빼기, 합침=False):
    """복사 → 지문 대조. 실패하면 새로 만든 것을 치우고 멈춤. 원래 자리는 지우지 않는다(원래지우기)."""
    확인(src, dst, 합침)
    목록 = 훑기(src, 빼기)
    만든 = []
    try:
        if 목록 and 목록[0][0] == ".":
            os.makedirs(os.path.dirname(dst), exist_ok=True)
        elif not os.path.isdir(dst):
            os.makedirs(dst)
            만든.append(dst)
        for rel, 종류 in 목록:
            s, d = 자리(src, rel), 자리(dst, rel)
            if 종류 == "폴더":
                if not os.path.isdir(d):
                    os.mkdir(d)
                    만든.append(d)
                continue
            if os.path.lexists(d):
                raise 멈춤(f"새 자리에 같은 이름이 있다: {d}")
            if 종류 == "링크":
                os.symlink(os.readlink(s), d)
            else:
                shutil.copy2(s, d)
            만든.append(d)
        다름 = [rel for rel, 종류 in 목록 if 지문(자리(src, rel), 종류) != 지문(자리(dst, rel), 종류)]
        if 다름:
            raise 멈춤(f"지문이 다르다 {len(다름)}개: {다름[:5]}")
    except (멈춤, OSError) as e:
        for p in reversed(만든):
            try:
                os.rmdir(p) if os.path.isdir(p) and not os.path.islink(p) else os.unlink(p)
            except OSError:
                pass
        raise 멈춤(f"{e} — 원래 자리는 그대로 · 새로 만든 것은 치웠다") from None
    n, b = 세기(dst, 목록)
    return n, b, 목록


def 원래지우기(src, 목록, 빼기):
    for rel, 종류 in 목록:
        if 종류 != "폴더":
            os.unlink(자리(src, rel))
    for rel, _ in sorted((x for x in 목록 if x[1] == "폴더"), key=lambda x: -x[0].count(os.sep)):
        try:
            os.rmdir(자리(src, rel))
        except OSError:
            pass  # 빼기가 든 폴더는 비지 않아 남는다
    if os.path.isdir(src) and not os.path.islink(src) and not 빼기:
        os.rmdir(src)


def 적기(방식, src, dst, 빼기, n, b):
    처음 = not 기록.exists()
    번호 = 1 if 처음 else sum(1 for _ in open(기록, encoding="utf-8"))
    with open(기록, "a", encoding="utf-8") as f:
        if 처음:
            f.write("\t".join(열) + "\n")
        f.write("\t".join(map(str, [번호, time.strftime("%Y-%m-%dT%H:%M:%S"), 방식, src, dst, ",".join(빼기), n, b])) + "\n")
    return 번호


def 하나(방식, src, dst, 빼기, 표시=None):
    src, dst = str(src), str(dst)
    if 방식 == "이름바꾸기":
        n, b = 이름바꾸기(src, dst, 빼기)
        return 적기(표시 or 방식, src, dst, 빼기, n, b), n, b
    if 방식 == "복사이동":
        n, b, 목록 = 복사이동(src, dst, 빼기)
        번호 = 적기(표시 or 방식, src, dst, 빼기, n, b)
        try:
            원래지우기(src, 목록, 빼기)
        except OSError as e:
            raise 멈춤(f"복사 · 대조는 끝남(기록 {번호}) · 원래 자리를 다 지우지 못함: {e}")
        return 번호, n, b
    raise 멈춤(f"모르는 방식: {방식}")


def 되돌리기(번호):
    with open(기록, encoding="utf-8") as f:
        줄 = [l.rstrip("\n").split("\t") for l in f][1:]
    r = next((x for x in 줄 if x[0] == str(번호)), None)
    if r is None:
        raise 멈춤(f"기록 {번호} 없음")
    방식, src, dst, 빼기 = r[2], r[3], r[4], [x for x in r[5].split(",") if x]
    if 방식 == "이름바꾸기":
        return 하나("이름바꾸기", dst, src, [], 표시=f"되돌리기({번호})")
    if 방식 == "복사이동":
        n, b, 목록 = 복사이동(dst, src, [], 합침=bool(빼기))
        새번호 = 적기(f"되돌리기({번호})", dst, src, [], n, b)
        원래지우기(dst, 목록, [])
        return 새번호, n, b
    raise 멈춤(f"되돌릴 수 없는 기록: {방식}")


def 목록실행(path):
    with open(path, encoding="utf-8") as f:
        줄들 = f.read().splitlines()
    for i, line in enumerate(줄들, 1):
        if not line.strip() or line.startswith("#"):
            continue
        p = line.split("\t")
        if len(p) < 3:
            print(f"🔴 {i}번째 줄 형식이 틀림: {line}")
            return 1
        빼기 = [x for x in (p[3].split(",") if len(p) > 3 else []) if x]
        try:
            번호, n, b = 하나(p[0], p[1], p[2], 빼기)
        except 멈춤 as e:
            print(f"🔴 {i}번째 줄에서 멈춤: {e}")
            return 1
        print(f"✅ [{번호}] {p[0]} · 파일 {n}개 {b / 1e6:.1f}MB · {p[1]} → {p[2]}")
    return 0


def main(argv=None):
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    sub = ap.add_subparsers(dest="일", required=True)
    sub.add_parser("목록").add_argument("파일")
    sub.add_parser("되돌리기").add_argument("번호", type=int)
    a = ap.parse_args(argv)
    if a.일 == "목록":
        return 목록실행(a.파일)
    try:
        번호, n, b = 되돌리기(a.번호)
    except 멈춤 as e:
        print(f"🔴 되돌리기 멈춤: {e}")
        return 1
    print(f"✅ [{번호}] 되돌림 · 파일 {n}개 {b / 1e6:.1f}MB")
    return 0


if __name__ == "__main__":
    sys.exit(main())
