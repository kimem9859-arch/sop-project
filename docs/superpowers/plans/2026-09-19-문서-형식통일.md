```python
import re
p = "/home/pi/sop-project/docs/성능검증-저널.md"
EMOJI = re.compile("[\U0001F000-\U0001FAFF←-⯿☀-➿️⃣]")

def clean(title):
    head, _, rest = title.partition(" ")          # "###"
    num, _, body = rest.partition(" ")            # "12.56"
    body = EMOJI.sub("", body)                    # 이모지 제거

    return "%s %s %s" % (head, num, body)

for l in open(p, encoding="utf-8").read().splitlines():
    if re.match(r"^### 12\.\d+ ", l):
        new = clean(l)
        if new != l:
            print("전: " + l)
            print("후: " + new)
            print()
```

> 🔑 **제목 텍스트를 보존한다** — `#### 결과` → `#### (1) 결과`. 기존 `-(N)` 참조 94쌍은 번호형 29절을 가리키므로 **건드리지 않는다.**

```python
import re
p = "/home/pi/sop-project/docs/성능검증-저널.md"
L = open(p, encoding="utf-8").read().splitlines()
idx = [i for i, l in enumerate(L) if re.match(r"^### 12\.\d+ ", l)]
for k, i in enumerate(idx):
    end = idx[k + 1] if k + 1 < len(idx) else len(L)
    hs = [(j, L[j]) for j in range(i, end) if L[j].startswith("#### ")]
    if not hs:

    if len(numbered) == len(hs):
```

```bash
print("번호형 %d · 서술형 %d · 혼재 %d · 없음 %d" % (num, des, mix, non))
```
