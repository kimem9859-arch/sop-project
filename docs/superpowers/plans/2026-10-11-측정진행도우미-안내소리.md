**Goal:** 측정하는 사람이 안경의 짧은 안내 소리와 노트북 한 화면만 보고 대본을 진행하고, NG 는 자동(안내와 다른 누름) · N 키(사람 · 장비)로만 남기며, 세기가 기록만으로 실제 판과 대본 판을 맞춰 센다 — 그 전제로 파이 TTS 를 Supertonic 2 로 바꾼다.

```python
"""런타임 TTS — LLM 이 만든 문장을 그때그때 합성한다 · 고정 소리(make_answers)도 이것으로 만든다.

정본: ../docs/superpowers/specs/2026-09-07-음성비서-LLM-design.md §4 · 엔진 교체 = 2026-10-11-측정진행도우미-안내소리-design.md D11

🔑 A 갈래의 사전 합성(make_answers.py)을 되돌리는 자리다. 문장이 매번 다르니
   미리 만들어 둘 수 없다.

🔴 **말하는 시간이 지연의 지배 항목이다** — LLM 문장은 A 갈래 답변(1.48초)의
   3.2~3.6배인 4.77~5.34초를 말한다(§10.62-(9)). 합성 자체는 0.4~1.05초다.

⚠️ 모델은 상주시킨다 — 적재가 약 0.93초라 매번 올리면 그만큼 늘어난다(§10.49).
🔑 엔진 = config.TTS_ENGINE — supertonic2(기본 · F1 · 4단계 · 1.25 · 22050Hz 로 낮춤) · kss(되돌리기).
"""
import array
import os
import re
import struct

import config

_D = os.path.expanduser("~/env/tts/vits-mimic3-ko_KO-kss_low")
# 🔑 영문 버튼 이름은 한글 읽기로 — Supertonic 은 영문을 영어로 읽는다(저널 §12.94-(8))
_READ = {"EMO": "비상정지", "B1": "비원", "B2": "비투", "B3": "비쓰리", "B4": "비포"}
_READ_RE = re.compile("|".join(_READ))
TRIM_REL = 0.02         # 앞뒤 무음 = 최고점의 2% 미만(저널 §12.94 말소리 길이 기준과 같다)
TRIM_PAD_SEC = 0.02     # 자른 뒤 남기는 여유


def read_out(text):
    """버튼 이름(B1~B4 · EMO)을 한글 읽기로 — 모든 합성이 지난다(고정 wav · LLM 답 · 안내)."""
    return _READ_RE.sub(lambda m: _READ[m.group(0)], text or "")


def downsample(x, src, dst):
    """정수배로 레이트를 낮춘다 — 창 씌운 sinc 저역 거름(새 나이퀴스트의 90%) 뒤 솎기."""
    import numpy as np
    x = np.asarray(x, dtype=np.float32).reshape(-1)
    if src == dst:
        return x
    if src % dst:
        raise ValueError(f"정수배가 아니다: {src} → {dst}")
    f = src // dst
    n = 63
    m = np.arange(n) - (n - 1) / 2
    fc = 0.45 / f
    h = 2 * fc * np.sinc(2 * fc * m) * np.hamming(n)
    h /= h.sum()
    return np.convolve(x, h, mode="same")[::f].astype(np.float32)


def trim(x, rate):
    """앞뒤 무음을 자른다 — 안내 소리의 「재생 시작」이 들리기 시작한 때와 같아야 한다(설계 D14)."""
    import numpy as np
    a = np.abs(np.asarray(x, dtype=np.float32))
    if a.size == 0 or a.max() <= 0:
        return np.asarray(x, dtype=np.float32)
    on = np.nonzero(a >= a.max() * TRIM_REL)[0]
    pad = int(rate * TRIM_PAD_SEC)
    return np.asarray(x, dtype=np.float32)[max(0, on[0] - pad): min(a.size, on[-1] + 1 + pad)]


def to_pcm(x):
    import numpy as np
    return (np.clip(np.asarray(x, dtype=np.float32), -1.0, 1.0) * 32767).astype("<i2").tobytes()


class Tts:
    """합성기 — 만들 때 모델을 올리고 계속 들고 있는다. engine = "supertonic2" · "kss"(없으면 config)."""

    def __init__(self, model_dir=None, threads=2, engine=None):
        self.engine = engine or config.TTS_ENGINE
        if self.engine == "kss":
            import sherpa_onnx
            d = model_dir or _D
            self._tts = sherpa_onnx.OfflineTts(sherpa_onnx.OfflineTtsConfig(
                model=sherpa_onnx.OfflineTtsModelConfig(
                    vits=sherpa_onnx.OfflineTtsVitsModelConfig(
                        model=f"{d}/ko_KO-kss_low.onnx",
                        tokens=f"{d}/tokens.txt",
                        data_dir=f"{d}/espeak-ng-data"),
                    num_threads=threads, provider="cpu"),
                max_num_sentences=1))
        elif self.engine == "supertonic2":
            from supertonic import TTS
            self._st = TTS(model="supertonic-2", model_dir=model_dir or config.TTS_ST_DIR, auto_download=False,
                           intra_op_num_threads=threads, inter_op_num_threads=1)
            self._style = self._st.get_voice_style(voice_name=config.TTS_ST_VOICE)
        else:
            raise ValueError(f"알 수 없는 TTS_ENGINE: {self.engine!r}(\"supertonic2\" 또는 \"kss\")")

    def synth(self, text, speed=None):
        """`(pcm bytes, 레이트, 말하는 시간초)` — 실패하면 None. speed 없음 = 엔진 기본(kss 1.0 · supertonic2 config)."""
        text = read_out(text)
        if not text.strip():
            return None
        try:
            if self.engine == "kss":
                a = self._tts.generate(text, sid=0, speed=1.0 if speed is None else speed)
                pcm = array.array("h", [int(max(-1.0, min(1.0, x)) * 32767) for x in a.samples])
                if not pcm:
                    return None
                return pcm.tobytes(), a.sample_rate, len(pcm) / a.sample_rate
            wav, _ = self._st.synthesize(text, voice_style=self._style, lang="ko", total_steps=config.TTS_ST_STEPS,
                                         speed=config.TTS_ST_SPEED if speed is None else speed)
            x = trim(downsample(wav, self._st.sample_rate, config.TTS_OUT_RATE), config.TTS_OUT_RATE)
            if x.size == 0:
                return None
            return to_pcm(x), config.TTS_OUT_RATE, x.size / config.TTS_OUT_RATE
        except Exception:                      # noqa: BLE001 — 데몬은 살아야 한다
            return None
```

🔴 순수 모듈이다(Qt · 장치 없음) — 사건 목록과 대본 줄만 받는다. 판정을 다시 하지 않고 시연이 적은 사건을 대조한다.

누름 칸 낱말: B2(맞는 누름) · B3@B1(B1 차례에 B3) · EMO@B4 · ~B4@B3(머묾 — 안내만) · #렌치(공구 — 안내만)
             · 끝의 ? = 그 낱말부터는 판이 먼저 끝나도 된다.
판 상태: ok 대본대로 · ng 자동(안내와 다른 버튼) 또는 수동(N 키) · partial 일찍 끝남(결과로 남김 · D16)
        · extra 대본을 다 끝낸 뒤 · open 아직 안 끝남(진행 도우미).
🔴 자동 NG 는 「안내와 다른 버튼이 실제로 눌림」 하나뿐 — 모델·판정 이상은 NG 가 아니다(D4).
"""
import csv
import os
import re
from collections import namedtuple

ACTIONS = ("정상", "위반", "머묾", "틀린공구", "호출", "질문", "비상질문")   # 대본 행동(세기 설계 §5)
TOOLS = {"렌치": "wrench", "드라이버": "driver", "플라이어": "pliers"}
Token = namedtuple("Token", "kind name expected optional row")              # kind = press · hover · tool
_TOK = re.compile(r"^(?P<hover>~)?(?P<tool>#)?(?P<name>[^@?~#\s]+)(?:@(?P<exp>[^@?~#\s]+))?(?P<opt>\?)?$")


def parse_tokens(text, buttons, emo="EMO", row=None):
    """누름 칸 → [Token] · 틀린 낱말은 ValueError(무엇이 틀렸는지)."""
    out = []
    for w in (text or "").split():
        m = _TOK.match(w)
        if not m:
            raise ValueError(f"낱말 「{w}」 을(를) 읽을 수 없다")
        name, exp, opt = m.group("name"), m.group("exp"), bool(m.group("opt"))
        if m.group("tool"):
            if name not in TOOLS or exp or m.group("hover"):
                raise ValueError(f"공구 낱말 「{w}」 — #{' · #'.join(TOOLS)} 가운데 하나")
            out.append(Token("tool", name, None, opt, row))
            continue
        if name not in buttons and name != emo:
            raise ValueError(f"「{w}」 의 버튼 {name} 은(는) {'·'.join(buttons)}·{emo} 가운데 하나여야 한다")
        if exp is not None and exp not in buttons:
            raise ValueError(f"「{w}」 의 차례 {exp} 은(는) {'·'.join(buttons)} 가운데 하나여야 한다")
        if m.group("hover"):
            if exp is None or name == emo:
                raise ValueError(f"머묾 낱말 「{w}」 은(는) ~버튼@차례 꼴이다")
            out.append(Token("hover", name, exp, opt, row))
        else:
            out.append(Token("press", name, exp if exp is not None else (None if name == emo else name), opt, row))
    return out


def load_script(path, buttons=None, emo="EMO"):
    """대본 CSV(세기 설계 §5 + 진행 도우미 설계 §4.2 「누름」 칸) — 빈 줄·BOM 은 받는다 · 행동 철자·판 숫자·
    위반/머묾 대상·누름 낱말 오류는 줄 번호와 함께 멈춘다. buttons 를 주면 「낱말」 칸에 [Token] 을 둔다."""
    name, rows = os.path.basename(path), []
    with open(path, encoding="utf-8-sig", newline="") as f:
        rd = csv.DictReader(f)
        for r in rd:
            n = rd.line_num                         # 빈 줄이 있어도 파일의 실제 줄 번호
            r = {k.strip(): (v or "").strip() for k, v in r.items() if k}
            if not any(r.values()):
                continue
            if r.get("행동") not in ACTIONS:
                raise ValueError(f"대본 {name} {n}행 — 행동 「{r.get('행동')}」 은(는) {'·'.join(ACTIONS)} 가운데 하나여야 한다")
            try:
                r["판"] = int(r.get("판", ""))
            except ValueError:
                raise ValueError(f"대본 {name} {n}행 — 판 「{r.get('판')}」 은(는) 1부터의 숫자여야 한다") from None
            for k in ("대상", "기대", "메모", "누름"):
                r.setdefault(k, "")
            if buttons and r["행동"] in ("위반", "머묾") and r["대상"] not in buttons:
                raise ValueError(f"대본 {name} {n}행 — {r['행동']} 대상 「{r['대상']}」 은(는) 레시피 버튼({'·'.join(buttons)}) 가운데 하나여야 한다")
            if buttons:
                try:
                    r["낱말"] = parse_tokens(r["누름"], buttons, emo, row=len(rows))
                except ValueError as e:
                    raise ValueError(f"대본 {name} {n}행 — 누름 칸: {e}") from None
            rows.append(r)
    return rows


def has_tokens(rows):
    """누름 칸이 하나라도 채워져 있나 — 없으면 세기는 판 맞춤을 건너뛰고 지금처럼 센다(되돌리기 · 설계 §9)."""
    return any(r.get("낱말") for r in rows or ())


def script_order(rows):
    """{대본 판: 그 판 낱말 전부(줄 차례대로)} — 판 번호 작은 것부터."""
    out = {}
    for r in rows or ():
        out.setdefault(int(r["판"]), []).extend(r.get("낱말") or ())
    return dict(sorted(out.items()))


def runs_of(events):
    """판 = run_start ~ run_end · run_reset — measure_count.split_runs 와 같은 규칙.
    끝 사건 없이 다음 run_start 가 오면 그 앞에서 닫는다(「끊김」) · 마지막 열린 판 = open."""
    out, cur = [], None
    for t, k, d in events:
        if k == "run_start":
            if cur is not None:
                cur.update(t1=t - 0.001, end="끊김")
                out.append(cur)
            cur = {"t0": t, "t1": None, "end": "open", "ev": []}
        elif cur is None:
            continue
        elif k in ("run_end", "run_reset"):
            cur.update(t1=t, end="완주" if k == "run_end" else "초기화")
            out.append(cur)
            cur = None
        else:
            cur["ev"].append((t, k, d))
    if cur is not None:
        out.append(cur)
    return out


def run_presses(ev, emo="EMO"):
    """판 안 누름 가운데 시연이 받은 것만 → [(t, 버튼, 기대)].
    무시 = 차단 중 EMO 아닌 버튼 · 진행 중 서브 작업 버튼 재입력(경고 중 정답 포함 — 단계를 올리지 않는다 · 설계 D4)."""
    sub, out = None, []
    for t, k, d in ev:
        if k == "sub":
            w = d.get("what")
            if w == "start":
                sub = d.get("button")
            elif w in ("finish", "cancel"):
                sub = None
        elif k == "press":
            b = d.get("button")
            if d.get("state") == "BLOCK" and b != emo:
                continue
            if sub is not None and b == sub:
                continue
            out.append((t, b, d.get("expected")))
    return out


def _simulate(presses, toks):
    """누름 → 그 대본 판의 누름 낱말에 차례대로 대 본다."""
    req = [x for x in toks if x.kind == "press"]
    first_opt = next((n for n, x in enumerate(req) if x.optional), len(req))
    i = 0
    for n, (_, b, e) in enumerate(presses):
        if i < len(req) and req[i].name == b and (req[i].expected is None or req[i].expected == e):
            i += 1
            continue
        return {"ok": False, "at": n, "got": (b, e), "want": req[i] if i < len(req) else None, "consumed": i}
    return {"ok": True, "consumed": i, "missing": max(0, first_opt - i)}


def _ng_why(s):
    b, e = s["got"]
    want = s["want"]
    w = "대본에 없는 누름" if want is None else f"안내 {want.name}@{want.expected or '아무 차례'}"
    return f"{s['at'] + 1}번째 누름 {b}@{e} · {w}"


def _owner(runs, t):
    """그 시각의 판 자리 — 판 안이면 그 판 · 판 사이면 막 끝난 판 · 첫 판 전이면 None."""
    last = None
    for i, r in enumerate(runs):
        if r["t0"] <= t and (r["t1"] is None or t <= r["t1"]):
            return i
        if r["t1"] is not None and r["t1"] < t:
            last = i
    return last


def match(events, rows, marks=(), ahead=0, emo="EMO"):
    """사건 → 판마다 결과 + 다음 대본 판.

    marks = [(t_ms, "ng"|"undo", 이유)] — 시각 차례로 쌓고 undo 는 마지막 ng 를 지운다 · ng 는 그 시각의 판에.
    ahead = 끝난 판을 지금 대본 판 → 다음 ahead 판까지 대 본다(세기 2 · 진행 도우미 0 — 설계 §4.3 「앞 찾기」).
    🔴 뒤 대본 판으로 건너뛰는 것은 그 판과 **끝까지 맞을 때만**(ok) — 앞부분만 같은 NG 시도를 건너뜀으로 읽지 않게.
       지금 대본 판은 미완(partial)도 받는다(D16).
    """
    order = script_order(rows)
    keys = list(order)
    ng = []
    for t, kind, why in sorted(marks, key=lambda m: m[0]):
        if kind == "ng":
            ng.append((t, why or "사람"))
        elif kind == "undo" and ng:
            ng.pop()
    runs = runs_of(events)
    manual = {}
    for t, why in ng:
        i = _owner(runs, t)
        if i is not None:
            manual[i] = why
    out, k = [], 0
    for i, r in enumerate(runs):
        res = {"i": i + 1, "t0": r["t0"], "t1": r["t1"], "end": r["end"], "script": None, "status": "", "why": "",
               "skipped": [], "missing": 0, "consumed": 0, "live_ng": False,
               "warn": sum(1 for _, kk, d in r["ev"] if kk == "state" and d.get("new") == "WARNING")}
        ps = run_presses(r["ev"], emo)
        if k >= len(keys):
            res.update(status="extra", why="대본을 다 끝낸 뒤의 판")
        elif r["end"] == "open":
            s = _simulate(ps, order[keys[k]])
            res.update(script=keys[k], status="open", consumed=s["consumed"])
            if not s["ok"]:
                res.update(live_ng=True, why="자동 — " + _ng_why(s))
        else:
            chosen = None
            for j in range(k, min(k + ahead, len(keys) - 1) + 1):
                s = _simulate(ps, order[keys[j]])
                if s["ok"] and (j == k or not s["missing"]):
                    chosen = (j, s)
                    break
            if chosen is None:
                s = _simulate(ps, order[keys[k]])
                res.update(script=keys[k], status="ng", why="자동 — " + _ng_why(s), consumed=s["consumed"])
            else:
                j, s = chosen
                res.update(script=keys[j], status="partial" if s["missing"] else "ok", skipped=keys[k:j],
                           missing=s["missing"], consumed=s["consumed"])
                k = j + 1
        if i in manual and res["status"] != "extra":
            if res["status"] in ("ok", "partial"):
                k = keys.index(res["script"])          # 다시 할 대본 판으로 되돌린다
            res.update(status="ng", why=f"수동 — {manual[i]}" + (f" · {res['why']}" if res["why"] else ""))
        out.append(res)
    return {"runs": out, "next": keys[k] if k < len(keys) else None, "order": order}
```

def split_guide(S, res):
    """판 맞춤(measure_script.match) → NG·대본 밖 판의 모든 기록(events·voice·frames·fsm)을 그 판째 빼고 대본을 남은 판
    차례로 다시 번호 매긴다 — 남은 판 i = 대본 판 i 라 값 함수는 그대로 쓴다(진행 도우미 설계 D12).
    🔑 빠진 판은 breaks 로 남겨 그 구간을 넘는 간격을 끊김으로 세지 않는다(split_recording 과 같은 방식)."""
    drop = [(r["t0"], r["t1"] if r["t1"] is not None else float("inf"))
            for r in res["runs"] if r["status"] in ("ng", "extra")]
    keep = [r for r in res["runs"] if r["status"] not in ("ng", "extra")]
    inside = lambda t: any(a <= t <= b for a, b in drop)           # noqa: E731
    lists = {"events": lambda e: e[0], "voice": lambda e: e[0], "frames": lambda r: r["t"], "fsm": lambda r: r["t"]}
    script = S["script"]
    if script is not None:
        by = {}
        for row in script:
            by.setdefault(int(row["판"]), []).append(row)
        script = [dict(row, 판=n) for n, r in enumerate(keep, 1) if r["script"] is not None for row in by.get(r["script"], [])]
    return dict(S, script=script, guide=res, breaks=list(S.get("breaks") or []) + drop,
                **{k: [x for x in S[k] if not inside(f(x))] for k, f in lists.items()})
```

def v_prevent(S):
    """1 위반 사전 차단율 — 누를 때 판정기 상태 = WARNING.
    진단(성공으로 세지 않는다 · 진행 도우미 설계 D13①) — 경고가 없던 눌림을 누르는 순간(GPIO 엣지)까지의 판정기 줄로 가른다:
    none = 그 버튼 구역에 손 없음 · ring = 링에만(박스 안 줄 없음) · late = 박스 안에서 체류 문턱을 넘긴 줄을 누르기 전에
    받았는데 판정기가 누름보다 늦게 처리 · short = 체류 미달(문턱을 못 채움 — 문턱이 기록에 없으면 이쪽)."""
    thr = (S.get("settings") or {}).get("FSM_DWELL_THRESHOLD_SEC")
    F = S["fsm"]
    ft = [r["t"] for r in F]
    out = {"n": 0, "k": 0, "late": 0, "short": 0, "ring": 0, "none": 0}
    for p in violations(S):
        out["n"] += 1
        if p["state"] == "WARNING":
            out["k"] += 1
            continue
        i = bisect.bisect_right(ft, p["t_at"]) - 1
        if i < 0 or F[i]["roi"] != p["button"]:
            out["none"] += 1
            continue
        j = i
        while j > 0 and F[j - 1]["roi"] == p["button"]:
            j -= 1
        stretch = F[j:i + 1]
        if not any(r["level"] == 2 for r in stretch):
            out["ring"] += 1
            continue
        hit = thr is not None and any(r["level"] == 2 and r["t"] - F[j]["t"] >= thr * 1000 for r in stretch)
        out["late" if hit else "short"] += 1
    return out
```

def _alert_plays(S):
    """알림 재생 사건만 — 안내(cue_) 재생은 뺀다 · 옛 기록은 key 가 없다(None = 그대로 쓴다)."""

def _alert_windows(S):
    """알림마다 (시각, 데이터, 다음 알림 시각) — 그 알림의 재생은 이 창 안에서만 찾는다(재생 하나를 둘이 나눠 쓰지 않게)."""
    al = [(t, d) for t, k, d in S["voice"] if k == "alert"]
    return [(t, d, al[n + 1][0] if n + 1 < len(al) else float("inf")) for n, (t, d) in enumerate(al)]


def _first_play(plays, t, t_next, key):
    """그 알림의 재생 시작 — 재생에 소리 키가 있으면 같은 키의 첫 재생(새 기록 · 뒤 알림이 먼저 와도 제 재생을 찾는다)
    · 키가 없으면 다음 알림 전의 첫 재생(옛 기록)."""
    return next(((x, dd) for x, k, dd in plays if k == "play_start" and x >= t
                 and (dd.get("key") == key if dd.get("key") is not None else x < t_next)), None)

        return {"delays": [], "missing": 0, "off": 1}

        if d.get("t_pub_ms") is None:

        hit = _first_play(plays, t, t_next, d.get("key"))
        if hit is None or hit[0] - t > ALERT_WAIT_MS:
            out["missing"] += 1
        else:
            out["delays"].append(round((hit[0] - d["t_pub_ms"]) / 1000, 3))

def v_alert_stop(S):
    """V3 알림 끊기 — 알림 재생 중에 해제(release ok)되면 → 다음 stop_sent(초). 재생 중이 아니었으면 세지 않는다."""
    plays = _alert_plays(S)
    starts = [t for t, k, _ in plays if k == "play_start"]
    ends = [t for t, k, _ in plays if k == "play_end"]
    stops = [t for t, k, _ in S["voice"] if k == "stop_sent"]
    out = {"delays": [], "missing": 0}
    for t, k, d in S["events"]:
        if k != "release" or not d.get("ok"):
            continue
        s = [x for x in starts if x <= t]
        if not s or any(s[-1] <= e <= t for e in ends):
            continue
        nxt = next((x for x in stops if x >= t), None)
        if nxt is None or nxt - t > ALERT_WAIT_MS:
            out["missing"] += 1
        else:
            out["delays"].append(round((nxt - t) / 1000, 3))
    return out


def v_alert_len(S):
    """V4 알림 길이 — 그 알림 창 안 첫 재생 시작 → 그 뒤 첫 재생 끝(초). 「중단」으로 끝난 재생은 길이가 아니라 cut."""
    plays = _alert_plays(S)
    out = {"lens": [], "cut": 0}
    for t, d, t_next in _alert_windows(S):
        hit = _first_play(plays, t, t_next, d.get("key"))
        if hit is None or hit[0] - t > ALERT_WAIT_MS:
            continue
        pe = next(((x, dd) for x, k, dd in plays if k == "play_end" and x >= hit[0]), None)
        if pe is None:
            continue
        if pe[1].get("what") == "중단":
            out["cut"] += 1
        else:
            out["lens"].append(round((pe[0] - hit[0]) / 1000, 3))
    return out

class CueWatcher:
    """측정 안내 소리(진행 도우미 설계 2026-10-11 D6) — 진행 도우미가 쓴 요청(req.json)의 새 번호를 보면 그 고정 wav 를 낸다.

    🔑 cue_ 로 시작하는 키만 · 3초보다 오래된 요청은 버린다(지난 세션에 남은 파일) · 끝나면 ack.json.
    🔑 알림과 따로 돈다 — 스피커(_io)가 차례를 정해 알림·답이 나가는 중이면 끝난 뒤에 낸다.
    `speaking` = 안내 재생 중 — 메인 루프가 그동안 들어온 소리를 버린다(알림과 같은 G11 원칙).
    """

    def __init__(self, spk, alog=None, d=None, poll_sec=ALERT_POLL_SEC, max_age_ms=3000.0):
        self.spk, self.alog = spk, alog
        self.d = d or config.CUE_SHM_DIR
        self.poll_sec, self.max_age_ms = poll_sec, max_age_ms
        self.speaking = threading.Event()
        self._seq = None

    def _read(self):
        try:
            with open(os.path.join(self.d, "req.json"), encoding="utf-8") as f:
                return json.load(f)
        except (OSError, ValueError):
            return None

    def step(self):
        """한 번 본다 — 낸 소리 키(없으면 None). 시험이 직접 부른다."""
        req = self._read()
        if not req or req.get("seq") == self._seq:
            return None
        self._seq = req.get("seq")
        key = str(req.get("key") or "")
        age = measure_log.now_ms() - float(req.get("t_req_ms") or 0)
        if not key.startswith("cue_") or age > self.max_age_ms:
            log(f"⚠️ 안내 요청을 버렸다 — {key or '(키 없음)'} · {age:.0f}ms 전")
            return None
        VLOG.event("cue", key=key, seq=self._seq, age_ms=round(age, 1))
        self.speaking.set()
        try:
            ok = self.spk.play(key, self.alog)
        finally:
            self.speaking.clear()
        self._ack(ok)
        log(f"📣 안내 → {key} · {'재생됨' if ok else '재생 확인 안 됨'}")
        return key

    def _ack(self, ok):
        try:
            os.makedirs(self.d, exist_ok=True)
            tmp = os.path.join(self.d, "ack.json.tmp")
            with open(tmp, "w", encoding="utf-8") as f:
                json.dump({"seq": self._seq, "ok": bool(ok), "t_ms": measure_log.now_ms()}, f)
            os.replace(tmp, os.path.join(self.d, "ack.json"))
        except OSError as e:
            log(f"⚠️ 안내 응답 파일을 못 썼다: {e}")

    def run(self, stop):
        while not stop.is_set():
            try:
                self.step()
            except Exception as e:              # noqa: BLE001 — 감시가 죽으면 안내가 영영 없다
                log(f"🔴 안내 감시 오류 — 계속한다: {type(e).__name__}: {e}")
            stop.wait(self.poll_sec)

    def start(self, stop):
        th = threading.Thread(target=self.run, args=(stop,), daemon=True)
        th.start()
        return th
```

#!/usr/bin/env python3
"""측정 진행 도우미 — 노트북 SSH 창에서 대본을 한 줄씩 보이고, 안경으로 짧은 안내 소리를 내고, NG 를 잡는다.

GAP_MS = 1000.0          # 직전 행동·안내에서 이만큼 지나야 다음 안내(설계 D8)
HOVER_WAIT_MS = 4000.0   # 머묾 안내 뒤 경고 해제가 없으면 이만큼 뒤 다음으로

def cue_key(tok):
    """낱말 → 고정 wav 키(make_answers.CUES)."""
    if tok.kind == "press":
        return f"cue_{tok.name}"
    if tok.kind == "hover":
        return f"cue_hover_{tok.name}"
    return f"cue_tool_{MS.TOOLS[tok.name]}"


def live_state(run):
    """열린 판 → 지금 판정기 상태 · 기대 버튼 · 마지막 행동 시각 · 경고 해제 시각들(설계 D8)."""
    st = {"state": None, "expected": None, "last": run["t0"], "released": []}
    for t, k, d in run["ev"]:
        if k == "state":
            st["state"], st["expected"] = d.get("new"), d.get("expected")
        if k in ("press", "release", "step_done", "sub"):
            st["last"] = t
        if k == "release" and d.get("what") == "warning" and d.get("ok"):
            st["released"].append(t)
    return st


def next_cue(toks, consumed, cued, st, now_ms):
    """지금 낼 안내 → (낱말 자리, 소리 키) 또는 None(설계 D8).
    toks = 그 대본 판의 낱말 전부 · consumed = 쓴 누름 낱말 수 · cued = {낱말 자리: 요청 시각 ms}."""
    presses = [n for n, x in enumerate(toks) if x.kind == "press"]
    start = presses[consumed - 1] + 1 if consumed else 0
    for n in range(start, len(toks)):
        x = toks[n]
        if n in cued:
            if x.kind == "press":
                return None                              # 낸 누름이 아직 안 눌렸다 — 기다린다
            if x.kind == "hover" and not (any(t >= cued[n] for t in st["released"])
                                          or now_ms - cued[n] >= HOVER_WAIT_MS):
                return None                              # 머묾 중 — 경고 해제 또는 4초까지
            continue
        if st["state"] not in RUN_STATES:
            return None
        if x.kind in ("press", "hover") and x.expected is not None and st["expected"] != x.expected:
            return None
        if now_ms - max([st["last"]] + list(cued.values())) < GAP_MS:
            return None
        return n, cue_key(x)
    return None


class Guide:
    """한 세션을 따라간다 — 판 맞춤 · 안내 결정 · NG 기록(소리·파일 쓰기는 sink 로 · 시험은 sink 를 바꾼다)."""

    def __init__(self, rows, emo, cue_sink=None, log_sink=None):
        self.rows, self.emo = rows or [], emo
        self.marks = []          # [(t, "ng"|"undo", 이유)]
        self.cued = {}           # {(실제 판 번호, 낱말 자리): 요청 시각}
        self.ng_cued = set()     # 「다시」를 낸 판
        self.cue_sink = cue_sink or (lambda key, t: None)
        self.log_sink = log_sink or (lambda t, kind, data: None)
        self.view = {"res": None, "run": None, "cur": None}
        self._n = -1
        self._res = None

    def restore(self, guide_rows):
        """guide.csv 를 다시 읽어 이어간다 — 낸 안내 · N/U(Review Focus 1)."""
        for t, k, d in guide_rows:
            if k == "cue" and d.get("tok") is not None:
                self.cued[(d.get("run"), d["tok"])] = t
            elif k == "cue" and d.get("key") == "cue_ng":
                self.ng_cued.add(d.get("run"))
            elif k in ("ng", "undo"):
                self.marks.append((t, k, d.get("why")))

    def mark(self, now_ms, kind, why=None):
        self.marks.append((now_ms, kind, why))
        self._n = -1                                     # 다시 맞춘다
        self.log_sink(now_ms, kind, {"why": why} if why else {})

    def step(self, events, now_ms):
        """→ ("cue", 키) · ("ng", "cue_ng") · None."""
        if not self.rows:
            self.view = {"res": None, "run": None, "cur": None}
            return None
        if len(events) != self._n:                       # 새 사건이 있을 때만 다시 맞춘다(CPU)
            self._res = MS.match(events, self.rows, self.marks, ahead=0, emo=self.emo)
            self._runs = MS.runs_of(events)
            self._n = len(events)
        res, runs = self._res, self._runs
        cur = runs[-1] if runs and runs[-1]["end"] == "open" else None
        r = res["runs"][-1] if cur is not None else None
        self.view = {"res": res, "run": r, "cur": cur}
        if r is None or r["script"] is None:
            return None
        if r["status"] == "ng" or r["live_ng"]:
            if r["i"] not in self.ng_cued and r["live_ng"]:
                self.ng_cued.add(r["i"])
                self.cue_sink("cue_ng", now_ms)
                self.log_sink(now_ms, "cue", {"run": r["i"], "script": r["script"], "key": "cue_ng"})
                return ("ng", "cue_ng")
            return None
        toks = res["order"][r["script"]]
        mine = {n: t for (i, n), t in self.cued.items() if i == r["i"]}
        c = next_cue(toks, r["consumed"], mine, live_state(cur), now_ms)
        if c is None:
            return None
        self.cued[(r["i"], c[0])] = now_ms
        self.cue_sink(c[1], now_ms)
        self.log_sink(now_ms, "cue", {"run": r["i"], "script": r["script"], "tok": c[0], "key": c[1]})
        return ("cue", c[1])


class Tail:
    """CSV 사건 파일 꼬리 읽기 — 새로 쓰인 완전한 줄만(기록기는 0.5초마다 내보낸다 · 처음부터 다시 읽지 않아 CPU 를 아낀다)."""

    def __init__(self, path):
        self.path, self.pos, self.rows, self._head = path, 0, [], False

    def poll(self):
        try:
            with open(self.path, "rb") as f:
                f.seek(self.pos)
                chunk = f.read()
        except OSError:
            return False
        end = chunk.rfind(b"\n")
        if end < 0:
            return False
        self.pos += end + 1
        new = False
        for line in csv.reader(io.StringIO(chunk[:end + 1].decode("utf-8"))):
            if not self._head:
                self._head = True
                continue
            if len(line) < 3:
                continue
            try:
                self.rows.append((float(line[0]), line[1], json.loads(line[2] or "{}")))
                new = True
            except ValueError:
                continue
        return new


def write_req(key, t, d=None):
    """안내 요청 — 음성비서 CueWatcher 가 읽는다(번호 = 요청 시각 ms · 다시 켜도 겹치지 않는다)."""
    d = d or config.CUE_SHM_DIR
    os.makedirs(d, exist_ok=True)
    tmp = os.path.join(d, "req.json.tmp")
    with open(tmp, "w", encoding="utf-8") as f:
        json.dump({"seq": int(t), "key": key, "t_req_ms": t}, f)
    os.replace(tmp, os.path.join(d, "req.json"))


def ack_status(last_req, d=None, now_ms=None):
    """마지막 요청 (번호, 시각) 에 음성비서가 답했나 — 화면 한 줄(Review Focus 2)."""
    if last_req is None:
        return "🔈 안내 전"
    try:
        with open(os.path.join(d or config.CUE_SHM_DIR, "ack.json"), encoding="utf-8") as f:
            ack = json.load(f)
    except (OSError, ValueError):
        ack = {}
    now_ms = measure_log.now_ms() if now_ms is None else now_ms
    if ack.get("seq") == last_req[0]:
        return "🔊 응답" if ack.get("ok") else "🔇 재생 확인 안 됨(안경 스피커 확인)"
    if now_ms - last_req[1] > ACK_WAIT_MS:
        return "🔇 응답 없음 — 음성비서 확인(화면 안내로 계속)"
    return "🔈 보내는 중"


class GuideLog:
    """세션 폴더 guide.csv — t_ms,kind,data(측정 기록과 같은 꼴 · 세기가 measure_check.load_events 로 읽는다)."""

    def __init__(self, path):
        self.path = path
        if not os.path.exists(path):
            with open(path, "w", encoding="utf-8", newline="") as f:
                csv.writer(f).writerow(["t_ms", "kind", "data"])

    def write(self, t, kind, data):
        with open(self.path, "a", encoding="utf-8", newline="") as f:
            csv.writer(f).writerow([round(t, 3), kind, json.dumps(data, ensure_ascii=False)])


def session_rows(d):
    """세션 폴더 → (session.json, 대본 줄 또는 None)."""
    with open(os.path.join(d, "session.json"), encoding="utf-8") as f:
        info = json.load(f)
    name = (info.get("입력") or {}).get("대본")
    if not name:
        return info, None
    return info, MS.load_script(os.path.join(d, name), MR.recipe_buttons(), MR.emo_button())


def newest_session(base=MEASURE_DIR):
    names = sorted(n for n in os.listdir(base)
                   if n not in SKIP_DIRS and os.path.isfile(os.path.join(base, n, "session.json")))
    return os.path.join(base, names[-1]) if names else None


def render(view, rows, info, ack, pending, name):
    """화면 글 — 바뀔 때만 다시 그린다."""
    res, r, cur = view["res"], view["run"], view["cur"]
    L = [f"{BOLD}SOP 가디언 측정 진행 — {name}{RST}    {ack}", ""]
    if not rows:
        L += ["대본 없음 — 안내 소리 없이 N 키만 쓴다", ""]
    elif res is not None:
        total = len(res["order"])
        ng = [x for x in res["runs"] if x["status"] == "ng"]
        k = r["script"] if r is not None else res["next"]
        L.append(f"{BOLD}대본 {k or '끝'}판 / 전체 {total}{RST} · 실제 {len(res['runs'])}번째 판 · NG {len(ng)}")
        if r is None:
            L.append(f"{GRN}▶ 「작업 시작」 → 대본 {k}판{RST}" if k else f"{GRN}대본을 다 했다 — ESC 로 시연 종료{RST}")
        show = k
        toks = res["order"].get(show, []) if show else []
        done = r["consumed"] if r is not None else 0
        presses_done = 0
        for row in [x for x in rows if x["판"] == show]:
            ws = []
            for tk in row.get("낱말") or ():
                mark = ""
                if tk.kind == "press":
                    mark = "✔" if presses_done < done else ""
                    presses_done += 1
                ws.append(f"{mark}{'~' if tk.kind == 'hover' else '#' if tk.kind == 'tool' else ''}{tk.name}"
                          + (f"@{tk.expected}" if tk.expected and tk.kind != "tool" and tk.expected != tk.name else ""))
            L.append(f"  {row['행동']} {row['대상']} — {row['메모']}")
            if ws:
                L.append(f"    {DIM}{' · '.join(ws)}{RST}")
        if r is not None and (r["live_ng"] or r["status"] == "ng"):
            L += ["", f"{RED}  NG — 「작업 초기화」 뒤 대본 {r['script']}판 다시  {RST}", f"  {r['why']}"]
        if cur is not None:
            last = [e for e in cur["ev"] if e[1] in ("press", "release", "state", "sub")]
            if last:
                t, k2, d = last[-1]
                L.append(f"\n  마지막: {k2} {d.get('button') or d.get('new') or d.get('what') or ''}")
            if any(k3 == "state" and d3.get("new") == "WARNING" for _, k3, d3 in cur["ev"][-3:]):
                L.append(f"  {YEL} 경고 — 대본에 없으면 결과로 기록된다(NG 아님) {RST}")
    if pending is not None:
        L += ["", f"{YEL} NG 이유: 1 사람 · 2 장비 (3초 · 안 고르면 사람) {RST}"]
    L += ["", f"{DIM}N = NG(사람·장비 때만 · 모델 이상은 누르지 않는다) · U = 취소 · Q = 끝{RST}"]
    return "\n".join(L)


def read_key(timeout):
    r, _, _ = select.select([sys.stdin], [], [], timeout)
    return sys.stdin.read(1) if r else None


def follow(d):
    """한 세션을 끝(measure_end)까지 따라간다 → "q"(끝) 또는 "next"."""
    info, rows = session_rows(d)
    gpath = os.path.join(d, "guide.csv")
    old = measure_check.load_events(gpath)
    glog = GuideLog(gpath)
    last_req = [None]

    def sink(key, t):
        write_req(key, t)
        last_req[0] = (int(t), t)
    g = Guide(rows, MR.emo_button(), cue_sink=sink, log_sink=glog.write)
    g.restore(old)
    tail = Tail(os.path.join(d, "events.csv"))
    pending, screen = None, None
    while True:
        tail.poll()
        now = measure_log.now_ms()
        g.step(tail.rows, now)
        key = read_key(TICK_SEC)
        if pending is not None and (key in ("1", "2") or now - pending >= REASON_MS):
            g.mark(pending, "ng", "장비" if key == "2" else "사람")
            pending, key = None, None
        if key in ("n", "N"):
            pending = now
        elif key in ("u", "U"):
            g.mark(now, "undo")
        elif key in ("q", "Q"):
            return "q"
        s = render(g.view, rows, info, ack_status(last_req[0]), pending, os.path.basename(d))
        if s != screen:
            sys.stdout.write("\033[2J\033[H" + s + "\n")
            sys.stdout.flush()
            screen = s
        if any(k == "measure_end" for _, k, _ in tail.rows[-3:]):
            sys.stdout.write("\n세션이 끝났다 — 다음 세션을 기다린다(Q = 끝)\n")
            sys.stdout.flush()
            return "next"


def run_live():
    import termios
    import tty
    fd = sys.stdin.fileno()
    old = termios.tcgetattr(fd)
    tty.setcbreak(fd)
    try:
        done = set()
        while True:
            d = newest_session()
            ended = d is None or d in done or any(
                k == "measure_end" for _, k, _ in measure_check.load_events(os.path.join(d, "events.csv")))
            if ended:
                sys.stdout.write("\033[2J\033[H측정 세션을 기다린다 — 실행기(SOP 가디언 측정)를 켜면 따라간다 · Q = 끝\n")
                sys.stdout.flush()
                if read_key(1.0) in ("q", "Q"):
                    return 0
                continue
            if follow(d) == "q":
                return 0
            done.add(d)
    finally:
        termios.tcsetattr(fd, termios.TCSADRAIN, old)


def replay(d, dump=True):
    """기록을 시간 순서로 다시 넣어 안내·NG 결정을 본다 — 소리·파일 쓰기 없음(설계 §6.3)."""
    info, rows = session_rows(d)
    ev = sorted(measure_check.load_events(os.path.join(d, "events.csv")), key=lambda e: e[0])
    g = Guide(rows, MR.emo_button())
    g.restore(measure_check.load_events(os.path.join(d, "guide.csv")))
    out, i, t = [], 0, ev[0][0] if ev else 0.0
    while i < len(ev):
        t += TICK_SEC * 1000
        while i < len(ev) and ev[i][0] <= t:
            i += 1
        c = g.step(ev[:i], t)
        if c:
            r = g.view["run"]
            out.append((round(t), r["i"], r["script"], c[1]))
            if dump:
                print(f"{t:14.1f} · {r['i']:2d}번째 판 · 대본 {r['script']} · {c[0]} {c[1]}")
    return out


def main(argv=None):
    ap = argparse.ArgumentParser(description="측정 진행 도우미(측정 진행 도우미 설계 2026-10-11)")
    ap.add_argument("--replay", default="", help="세션 폴더 — 기록을 다시 돌려 안내·NG 결정을 본다")
    ap.add_argument("--dump", action="store_true", help="--replay 결정을 한 줄씩 찍는다")
    a = ap.parse_args(argv)
    if a.replay:
        replay(a.replay, dump=True)
        return 0
    return run_live()


if __name__ == "__main__":
    sys.exit(main())
