```python

돌리기: cd ~/lab/storage-audit/20261016 && python3 -m unittest -v test_정돈
"""
import gzip, hashlib, os, re, shutil, tempfile, time, unittest
from pathlib import Path

import 정돈


def md5(p):
    return hashlib.md5(Path(p).read_bytes()).hexdigest()


class 정돈시험(unittest.TestCase):
    def setUp(self):
        # /tmp 아래에 만들면 도구가 tmp 안 사본을 「버림」으로 봐 사본 시험이 깨진다 — 이 폴더 안에 만든다
        self.t = Path(tempfile.mkdtemp(prefix="정돈시험-", dir=Path(__file__).resolve().parent))
        self.src, self.dst, self.desk = self.t / "출발", self.t / "받는곳", self.t / "데스크톱"
        (self.src / "가 나" / "깊은").mkdir(parents=True)

        (self.src / "빈파일.txt").write_bytes(b"")
        self.큰 = os.urandom(20000)
        (self.src / "큰것.bin").write_bytes(self.큰)
        (self.src / "휴지통것.bin").write_bytes(b"x" * 3000)
        os.symlink("/home/pi/어딘가/없는.png", self.src / "절대링크.png")
        os.symlink("가 나/사진 (1).png", self.src / "상대링크.png")
        # 데스크톱 사본 — 큰것.bin 은 쓸 수 있는 사본 · 휴지통것.bin 은 휴지통 안 사본(설계 규칙 5 — 치지 않음)
        (self.desk / "학습실험").mkdir(parents=True)
        (self.desk / "$RECYCLE.BIN").mkdir()
        (self.desk / "학습실험" / "큰것-사본.bin").write_bytes(self.큰)
        (self.desk / "$RECYCLE.BIN" / "휴지통것.bin").write_bytes(b"x" * 3000)
        self.후보 = self.t / "same_desk.tsv.gz"
        with gzip.open(self.후보, "wt", encoding="utf-8") as f:
            f.write(f"20000\tpi1:data/시험/큰것.bin\t{self.desk}/학습실험/큰것-사본.bin\n")
            f.write(f"3000\tpi1:data/시험/휴지통것.bin\t{self.desk}/$RECYCLE.BIN/휴지통것.bin\n")
        정돈.작업 = self.t / "작업"

    def 지문(self, root):
        out = {}
        for dp, dns, fns in os.walk(root):
            for n in fns + dns:
                p = Path(dp) / n
                rel = str(p.relative_to(root))
                if p.is_symlink():
                    out[rel] = ("링크", os.readlink(p))
                elif p.is_dir():
                    out[rel] = ("폴더",)
                else:
                    out[rel] = ("파일", md5(p))
```

```python
#!/usr/bin/env python3
"""작업/*/ 의 상태 · 목록으로 받은 쪽 안내문(README.md)을 만든다.

"""
import collections, json, sys
from pathlib import Path

작업 = Path(__file__).resolve().with_name("작업")
쪽 = sys.argv[1]
줄 = [f"# {'파이에서 옮겨 온 자료' if 쪽 == 'wsl-train' else '파이1 보관함'} — 2026-10-16 역할별 정돈", "",
      "> 설계 = sop-project `docs/superpowers/specs/2026-10-16-파이-데스크톱-역할별-파일정돈-design.md` · 도구 · 기록 원본 = 파이1 `~/lab/storage-audit/20261016/`",

기댐 = collections.Counter()
for d in sorted(작업.iterdir()):
    if not (d / "상태.json").exists():
        continue
    st = json.loads((d / "상태.json").read_text(encoding="utf-8"))
    if st["받는곳"][0] != 쪽 or "지우기" not in st["단계"]:
        continue
    with open(d / "목록.tsv", encoding="utf-8") as f:
        rows = [l.rstrip("\n").split("\t") for l in f][1:]
    파일 = [r for r in rows if r[1] != "폴더"]
    이미 = [r for r in 파일 if r[5] == "이미있음" and not r[6].startswith(st["받는곳"][1] + "/")]
    for r in 이미:
        기댐["/".join(r[6].split("/")[:5])] += 1
    줄.append(f"| {d.name} | `{st['출발'][0] or 'pi1'}:{st['출발'][1]}` | `{st['받는곳'][1]}` | {len(파일)} | "

if 기댐:

if 쪽 == "wsl-train":
```
