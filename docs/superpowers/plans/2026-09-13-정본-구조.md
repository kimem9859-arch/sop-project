import re
t=open("docs/통합문서.md",encoding="utf-8").read()

dup=[n for n in set(got) if got.count(n)>1]; miss=set(sec)-set(got); extra=set(got)-set(sec)
print("S3 ✅ %d/%d" % (len(set(got)),len(sec)) if not (dup or miss or extra) else "S3 ❌ 중복 %s 누락 %s 없는번호 %s" % (dup,sorted(miss),sorted(extra)))
