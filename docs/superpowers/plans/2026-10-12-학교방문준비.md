"""가짜 장치 — 카메라(형식 · 속도 · 램프 흉내) · 인터락(ACK) · 카메라 주소 덮기(학교 방문 준비 설계 §7.1).

실행: python3 Demo/selftest/test_fake_devices.py
"""
import os
import socket
import struct
import subprocess
import sys
import time

_DEMO_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, _DEMO_DIR)
sys.path.insert(0, os.path.join(_DEMO_DIR, "test"))

import cv2
import numpy as np

import fake_camera as FC
import fake_interlock as FI

_fails = []


def check(cond, msg):
    print(("  ✅ " if cond else "  ❌ ") + msg)
    if not cond:
        _fails.append(msg)


def _jpeg(v, w=160, h=120):
    return cv2.imencode(".jpg", np.full((h, w, 3), v, np.uint8))[1].tobytes()


def _read_one(s):
    n = struct.unpack("<I", _exact(s, 4))[0]
    return _exact(s, n)


def _exact(s, n):
    b = b""
    while len(b) < n:
        part = s.recv(n - len(b))
        if not part:
            raise ConnectionError("끊김")
        b += part
    return b


def test_camera_format_and_rate():
    print("\n[가짜 카메라] 안경과 같은 형식(4바이트 길이 + JPEG) · 정한 속도로 되풀이")
    frames = [_jpeg(10), _jpeg(200)]
    cam = FC.FakeCamera(frames, port=0, fps=40).start()
    try:
        s = socket.create_connection(("127.0.0.1", cam.port), timeout=3)
        got = [_read_one(s) for _ in range(3)]
        check(got == [frames[0], frames[1], frames[0]], "보낸 바이트 = 원본 그대로 · 차례대로 되풀이")
        t0, n0 = time.monotonic(), cam.sent
        time.sleep(1.0)
        rate = (cam.sent - n0) / (time.monotonic() - t0)
        check(30 <= rate <= 45, f"속도 약 40장/초 → {rate:.1f}")
        s.close()
    finally:
        cam.stop()


def test_camera_lamp_mimic():
    print("\n[가짜 카메라] 램프 흉내 — 정한 지연 뒤부터 램프 자리가 밝다")
    cam = FC.FakeCamera([_jpeg(60, 320, 240)], port=0, fps=40, mimic=True).start()
    try:
        s = socket.create_connection(("127.0.0.1", cam.port), timeout=3)
        x1, y1, x2, y2 = FC.LAMP_BOX

        def lamp_level():
            img = cv2.imdecode(np.frombuffer(_read_one(s), np.uint8), cv2.IMREAD_GRAYSCALE)
            return float(img[y1:y2, x1:x2].mean())

        check(lamp_level() < 40, "켜기 전 = 어둡다")
        cam.lamp(True, 0.2)
        t0 = time.monotonic()
        while lamp_level() < 200:
            if time.monotonic() - t0 > 2:
                break
        dt = time.monotonic() - t0
        check(0.15 <= dt <= 0.5, f"켜진 장이 지연 0.2초 뒤에 온다 → {dt:.2f}초")
        s.close()
    finally:
        cam.stop()


def test_interlock_ack():
    print("\n[가짜 인터락] 명령 한 줄 → ACK · 받은 명령 기록 · 콜백")
    import serial
    seen = []
    fi = FI.FakeInterlock(on_cmd=lambda cmd, t: seen.append(cmd)).start()
    try:
        ser = serial.Serial(fi.port, 115200, timeout=1)
        ser.write(b"WARN\n")
        ack = ser.readline()
        check(ack == b"ACK\n", f"ACK 응답 → {ack!r}")
        check([c for _, c in fi.cmds] == ["WARN"] and seen == ["WARN"], f"기록 {fi.cmds} · 콜백 {seen}")
        ser.close()
    finally:
        fi.stop()


def test_camera_host_override():
    print("\n[주소 덮기] SOP_CAMERA_HOST 가 있으면 시연 · 음성비서 둘 다 그 주소로(.camera_ip 는 그대로)")
    env = dict(os.environ, SOP_CAMERA_HOST="127.0.0.1")
    r = subprocess.run([sys.executable, "-c", "import config; print(config.CAMERA_TCP_HOST)"], cwd=_DEMO_DIR,
                       env=env, capture_output=True, text=True)
    check(r.stdout.strip() == "127.0.0.1", f"시연 = {r.stdout.strip()!r}")
    r = subprocess.run([sys.executable, "-c", "import voice_assistant as V; print(V.esp_ip())"], cwd=_DEMO_DIR,
                       env=env, capture_output=True, text=True)
    check(r.stdout.strip().splitlines()[-1:] == ["127.0.0.1"], f"음성비서 = {r.stdout.strip()[-40:]!r}")
    env.pop("SOP_CAMERA_HOST")
    r = subprocess.run([sys.executable, "-c", "import config; print(config.CAMERA_TCP_HOST)"], cwd=_DEMO_DIR,
                       env=env, capture_output=True, text=True)
    check(r.stdout.strip() != "127.0.0.1", f"없으면 종전(.camera_ip) = {r.stdout.strip()!r}")

설계 = 상위 sop-project docs/superpowers/specs/2026-10-12-학교방문준비-design.md §7.1 · §6(램프 흉내)

시험: from fake_camera import FakeCamera — port 0 이면 빈 포트를 고른다.

🔴 답하는 것 = 받기 · 풀기 · 처리 흐름이 끝까지 도는지 · 처리 fps(임시 값 — 장면 · JPEG 크기 · 무선이 실물과 다르다).
   답하지 않는 것 = 실제 안경의 송출 속도 · 지연 · 화질(학교 값이 본 값 · 설계 D16).
"""
import argparse
import glob
import os
import socket
import struct
import threading
import time

import cv2
import numpy as np

LAMP_BOX = (40, 40, 120, 120)   # 램프 흉내 — 밝게 칠할 자리(x1, y1, x2, y2 · 보내는 장 좌표)


def load_frames(src, limit=0):
    """폴더(f*.jpg — 안경이 보낸 바이트 그대로) 또는 영상(mp4 — 장마다 JPEG 로 다시 압축) → [JPEG 바이트]."""
    if os.path.isdir(src):
        out = []
        for p in sorted(glob.glob(os.path.join(src, "f*.jpg"))):
            with open(p, "rb") as f:
                out.append(f.read())
    else:
        cap = cv2.VideoCapture(src)
        out = []
        while not limit or len(out) < limit:
            ok, img = cap.read()
            if not ok:
                break
            out.append(cv2.imencode(".jpg", img, [cv2.IMWRITE_JPEG_QUALITY, 85])[1].tobytes())
        cap.release()
    out = out[:limit] if limit else out
    if not out:
        raise SystemExit(f"보낼 장이 없다: {src}")
    return out


class FakeCamera:
    """손님 하나에게 장을 정한 속도로 되풀이해 보낸다 — 끊으면 다음 손님을 기다린다(안경처럼 손님 하나)."""

    def __init__(self, frames, port=8888, fps=19.5, host="127.0.0.1", mimic=False):
        self.frames, self.fps, self.mimic = frames, fps, mimic
        self.sent = 0
        self._lamp_next = None        # (바뀔 시각 monotonic, 켬?)
        self._lamp_on = False
        self._stop = threading.Event()
        self._srv = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self._srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self._srv.bind((host, port))
        self._srv.listen(1)
        self._srv.settimeout(0.5)
        self.port = self._srv.getsockname()[1]

    def start(self):
        threading.Thread(target=self._loop, daemon=True).start()
        return self

    def stop(self):
        self._stop.set()
        try:
            self._srv.close()
        except OSError:
            pass

    def lamp(self, on, delay_s):
        """램프 흉내 — delay_s 뒤부터 보내는 장의 램프 자리를 밝게(on) · 어둡게(off) 칠한다(mimic=True 일 때)."""
        self._lamp_next = (time.monotonic() + delay_s, bool(on))

    def _frame(self, i):
        data = self.frames[i % len(self.frames)]
        if not self.mimic:
            return data
        nxt = self._lamp_next
        if nxt is not None and time.monotonic() >= nxt[0]:
            self._lamp_on, self._lamp_next = nxt[1], None
        # 🔑 켜든 끄든 매 장 풀고 다시 압축한다 — 램프 흉내 중 장마다 걸리는 시간이 같게
        img = cv2.imdecode(np.frombuffer(data, np.uint8), cv2.IMREAD_COLOR)
        x1, y1, x2, y2 = LAMP_BOX
        img[y1:y2, x1:x2] = 255 if self._lamp_on else 0
        return cv2.imencode(".jpg", img, [cv2.IMWRITE_JPEG_QUALITY, 85])[1].tobytes()

    def _loop(self):
        period = 1.0 / self.fps
        while not self._stop.is_set():
            try:
                c, _ = self._srv.accept()
            except socket.timeout:
                continue
            except OSError:
                return
            i, t_next = 0, time.monotonic()
            try:
                while not self._stop.is_set():
                    data = self._frame(i)
                    c.sendall(struct.pack("<I", len(data)) + data)
                    self.sent += 1
                    i += 1
                    t_next += period
                    time.sleep(max(0.0, t_next - time.monotonic()))
            except OSError:
                pass                                   # 손님이 끊음 — 다음 손님을 기다린다
            finally:
                c.close()


def main():
    ap = argparse.ArgumentParser(description="가짜 안경 카메라(학교 방문 준비 설계 §7.1)")
    ap.add_argument("src", help="f*.jpg 폴더 또는 mp4")
    ap.add_argument("--fps", type=float, default=19.5)
    ap.add_argument("--port", type=int, default=8888)
    ap.add_argument("--limit", type=int, default=0, help="앞에서 이만큼만(0 = 전부)")
    ap.add_argument("--seconds", type=float, default=0.0, help="이만큼 뒤 끝(0 = Ctrl+C 까지)")
    a = ap.parse_args()
    cam = FakeCamera(load_frames(a.src, a.limit), port=a.port, fps=a.fps).start()
    print(f"가짜 카메라 127.0.0.1:{cam.port} · {len(cam.frames)}장 · {a.fps}장/초", flush=True)
    t_end = time.monotonic() + a.seconds if a.seconds > 0 else None
    try:
        while t_end is None or time.monotonic() < t_end:
            time.sleep(0.2)
    except KeyboardInterrupt:
        pass
    cam.stop()
    print(f"보낸 장 {cam.sent}")

    main()

"""가짜 인터락 아두이노 — 가상 시리얼(pty)로 명령 한 줄(RUN · WARN · BLOCK)을 받아 「ACK」 로 답한다.

설계 = 상위 sop-project docs/superpowers/specs/2026-10-12-학교방문준비-design.md §7.1 · 규약 = interlock.py 머리말(명령 + \\n → ACK)
쓰는 법: fi = FakeInterlock(on_cmd=…).start() → SOP_INTERLOCK_PORT=fi.port 로 시연 · 램프 도구를 붙인다.
"""
import os
import pty
import threading
import time
import tty


class FakeInterlock:
    def __init__(self, on_cmd=None, ack=True):
        self.on_cmd = on_cmd or (lambda cmd, t: None)
        self.ack = ack
        self.cmds = []
        self._master, self._slave = pty.openpty()
        tty.setraw(self._slave)                  # 줄 바꿈 변환 없이(아두이노 시리얼처럼)
        self.port = os.ttyname(self._slave)      # 🔑 slave 를 열어 둔다 — 닫으면 상대가 열기 전에 사라진다
        self._stop = threading.Event()

    def start(self):
        threading.Thread(target=self._loop, daemon=True).start()
        return self

    def stop(self):
        self._stop.set()
        for fd in (self._master, self._slave):
            try:
                os.close(fd)
            except OSError:
                pass

    def _loop(self):
        buf = b""
        while not self._stop.is_set():
            try:
                chunk = os.read(self._master, 256)
            except OSError:
                return
            buf += chunk
            while b"\n" in buf:
                line, buf = buf.split(b"\n", 1)
                cmd = line.strip().decode("ascii", "replace")
                if not cmd:
                    continue
                t = time.monotonic()
                self.cmds.append((t, cmd))
                self.on_cmd(cmd, t)
                if self.ack:
                    try:
                        os.write(self._master, b"ACK\n")
                    except OSError:
                        return

"""가상 작업자 — 집 리허설에서 사람 대신 진행 도우미의 안내를 따라 시연을 누른다(학교 방문 준비 설계 §7.1).

main.py 가 SOP_REHEARSE_WORKER=1 일 때만 붙인다(시연 창 안 · QTimer 0.1초).
🔴 사람 흉내의 한계 — 손을 버튼 위에 올리는 「머묾」은 못 한다(카메라 장면이 녹화본이라) → 그 안내는 넘긴다(진행 도우미가 4초 뒤 다음으로).
"""
import csv
import io
import json
import os
import re
import time

import config
import measure_prompt
from fsm import State

REACT_MS = 800        # 안내를 듣고 누르기까지(사람 흉내)
START_GAP_MS = 1500   # 판 끝 → 다음 「작업 시작」
RELEASE_MS = 1500     # 경고 · 차단 배너 → 해제 버튼
END_IDLE_MS = 4000    # 판을 다 하고 → 창 닫기(세기가 돈다)
SAY_FILE = "rehearse_say.json"
SAY_TITLE = "🎤 지금 말하세요"


def action_for(cue_key):
    """진행 도우미 안내 키 → 할 일 · 머묾 · NG 안내는 None."""
    if cue_key == "cue_ng" or cue_key.startswith("cue_hover_"):
        return None
    if cue_key == "cue_reset":
        return ("reset",)
    if cue_key.startswith("cue_tool_"):
        return ("tool",)
    if cue_key.startswith("cue_"):
        return ("press", cue_key[4:])
    return None


def new_cues(path, pos):
    """guide.csv 의 pos(바이트) 뒤 **끝난 줄**에서 cue 키 → ([키], 새 pos) · 없으면 ([], pos)."""
    try:
        with open(path, "rb") as f:
            f.seek(pos)
            chunk = f.read()
    except OSError:
        return [], pos
    end = chunk.rfind(b"\n")
    if end < 0:
        return [], pos
    keys = []
    for row in csv.reader(io.StringIO(chunk[:end + 1].decode("utf-8"))):
        if len(row) < 3 or row[1] != "cue":
            continue
        try:
            k = json.loads(row[2]).get("key")
        except ValueError:
            continue
        if k:
            keys.append(k)
    return keys, pos + end + 1


def say_text(line):
    """말 띠 둘째 줄 「가디언」 → 「질문」 → 질문(마지막 꺾쇠) · 꺾쇠가 없으면 그대로."""
    found = re.findall(r"「([^」]*)」", line or "")
    return found[-1] if found else (line or "")


class Worker:
    def __init__(self, win, session_dir, runs_total, seconds=0, clock=None, shm_dir=None):
        self.win, self.runs_total, self.seconds = win, runs_total, seconds
        self.guide = os.path.join(session_dir, "guide.csv")
        self.shm = shm_dir or config.CUE_SHM_DIR
        self.clock = clock or (lambda: time.monotonic() * 1000)
        self._pos = 0
        self._due = []            # [(때 ms, 할 일)]
        self._started = 0
        self._ng = 0
        self._say_n = 0
        self._say_m = None
        self._idle_since = None
        self._alert_since = None
        self._t_first = None

    def _say(self):
        p = measure_prompt.read(self.shm)
        m = measure_prompt.mtime(self.shm)
        if p and p[0].startswith(SAY_TITLE) and m != self._say_m:
            self._say_m = m
            self._say_n += 1
            os.makedirs(self.shm, exist_ok=True)
            tmp = os.path.join(self.shm, SAY_FILE + ".tmp")
            with open(tmp, "w", encoding="utf-8") as f:
                json.dump({"n": self._say_n, "text": say_text(p[1])}, f, ensure_ascii=False)
            os.replace(tmp, os.path.join(self.shm, SAY_FILE))
        return p is not None

    def _do(self, a):
        w = self.win
        if a[0] == "press":
            w._press_button(a[1])
        elif a[0] == "tool":
            w._sim_tool_grasped()
        elif a[0] == "reset":
            w._reset_work()

    def tick(self):
        now, w = self.clock(), self.win
        keys, self._pos = new_cues(self.guide, self._pos)
        for k in keys:
            if k == "cue_ng":
                self._ng += 1                                   # 진행 도우미가 「다시」 — 판 하나를 더 한다
            a = action_for(k)
            if a:
                self._due.append((now + REACT_MS, a))
        band = self._say()
        for item in [x for x in self._due if x[0] <= now]:
            self._due.remove(item)
            self._do(item[1])
        mode = getattr(w.alert, "mode", None)
        if mode in ("block", "order"):
            self._alert_since = self._alert_since if self._alert_since is not None else now
            if now - self._alert_since >= RELEASE_MS:
                w._on_alert_release()
                self._alert_since = None
        else:
            self._alert_since = None
        if self.seconds and self._t_first is not None and now - self._t_first >= self.seconds * 1000:
            w.close()
            self.seconds = 0
            self.runs_total = -1                                # 다시 닫지 않게
            return
        if w.fsm.state != State.IDLE or self._due:
            self._idle_since = None
            return
        self._idle_since = self._idle_since if self._idle_since is not None else now
        idle = now - self._idle_since
        if not w.result_panel.isHidden():
            if idle >= START_GAP_MS:
                w._close_result()
                self._idle_since = now
            return
        want = (1 if self.seconds else self.runs_total + self._ng)
        if self._started < want and not band and idle >= START_GAP_MS:
            w._on_cta()
            self._started += 1
            self._t_first = self._t_first if self._t_first is not None else now
            self._idle_since = None
        elif self._started >= want and not self.seconds and self.runs_total >= 0 and idle >= END_IDLE_MS:
            w.close()
            self.runs_total = -1


def _runs_in_session(d):
    """세션 폴더의 대본 → 판 수(없으면 0)."""
    import measure_script as MS
    for n in sorted(os.listdir(d)):
        if n.endswith(".csv") and n not in ("frames.csv", "boxes.csv", "env.csv", "events.csv", "fsm.csv",
                                            "guide.csv", "voice_events.csv"):
            try:
                return len({r["판"] for r in MS.load_script(os.path.join(d, n))})
            except ValueError:
                return 0
    return 0


def attach(win):
    """시연 창에 붙인다 — SOP_MEASURE_DIR(세션) · SOP_REHEARSE_RUNS · SOP_REHEARSE_SECONDS."""
    from PyQt6.QtCore import QTimer
    d = os.environ.get("SOP_MEASURE_DIR") or ""
    runs = _runs_in_session(d) if d else 0
    cap = int(os.environ.get("SOP_REHEARSE_RUNS", "0") or 0)
    if cap:
        runs = min(runs, cap) if runs else cap
    wk = Worker(win, d, runs, seconds=float(os.environ.get("SOP_REHEARSE_SECONDS", "0") or 0))
    t = QTimer(win)
    t.timeout.connect(wk.tick)
    t.start(100)
    wk.timer = t
    print(f"[리허설] 가상 작업자 — 판 {runs} · 초 {wk.seconds}", flush=True)
    return wk

```python
class InferRunner:
    """모델 하나를 공유 장치에 InferModel 로 올린다 — 추론 중 GIL 을 놓는다(버림용 실험 · 학교 방문 준비 설계 §3.2)."""

    def __init__(self, hef_path):
        from contextlib import ExitStack
        import numpy as np
        from hailo_platform import FormatType
        import hailo_device
        self._lock = threading.Lock()
        self._stack = ExitStack()
        self._im = hailo_device.get_vdevice().create_infer_model(os.path.abspath(hef_path))
        self._im.input().set_format_type(FormatType.UINT8)
        for o in self._im.outputs:
            o.set_format_type(FormatType.FLOAT32)          # 지금 방식(InferVStreams FLOAT32)과 같은 역양자화 값
        self._cfg = self._stack.enter_context(self._im.configure())
        self._bind = self._cfg.create_bindings()
        self.input_name = self._im.input().name
        self._in_shape = tuple(self._im.input().shape)
        self._outs = {o.name: np.empty(o.shape, np.float32) for o in self._im.outputs}
        for name, buf in self._outs.items():
            self._bind.output(name).set_buffer(buf)
        self.output_names = list(self._outs)

    def infer(self, inputs):
        import numpy as np
        x = np.ascontiguousarray(next(iter(inputs.values())), dtype=np.uint8).reshape(self._in_shape)
        with self._lock:
            self._bind.input().set_buffer(x)
            self._cfg.run([self._bind], TIMEOUT_MS)
            return {k: v.copy()[np.newaxis] for k, v in self._outs.items()}   # 배치 축 — InferVStreams 결과와 같은 모양

    def close(self):
        self._stack.close()


def parse_nms_by_class(flat, n_cls):
    """NMS(클래스별) 평평한 출력 → 클래스마다 (k, 5) [ymin, xmin, ymax, xmax, score] — 클래스마다 개수 뒤에 상자가 이어진다."""
    import numpy as np
    flat = np.asarray(flat, np.float32).reshape(-1)
    out, k = [], 0
    for _ in range(n_cls):
        cnt = int(flat[k]) if k < flat.size else 0
        k += 1
        out.append(flat[k:k + cnt * 5].reshape(cnt, 5))
        k += cnt * 5
    return out
```
